"""Required #736 contract; run from repository root with python < this file.

This is intentionally retained outside test discovery because the required
contract is unresolved. A failing run is a blocker, never passing coverage.
"""
import inspect
import json
import logging
import random
from unittest.mock import patch
logging.basicConfig(level=logging.ERROR)
random.seed(736)
from actions import Action, Condition, State
from tiny_characters import Character, Goal
from tiny_gameplay_controller import GameplayController
from tiny_goap_system import GOAPPlanner, Plan
from tiny_graph_manager import GraphManager
from networkx import MultiDiGraph

graph = GraphManager()
controller = GameplayController(graph, {
    'headless': True, 'memory': {'mode': 'basic'},
    'characters': {'count': 1, 'use_llm': False, 'use_custom': True,
        'character_data': [{'name': 'ContractActor', 'age': 30,
            'job': 'unemployed', 'energy': 20}]},
    'checkpoint': {'auto_enabled': False}})
try:
    character = next(c for c in controller.characters.values() if c.name == 'ContractActor')
    assert isinstance(character, Character) and character._initialized
    assert isinstance(graph.G, MultiDiGraph)
    assert graph.characters[character.name] is character
    assert any(node is character for node in graph.G)
    planner = GOAPPlanner(graph)
    condition = Condition('energy_target', 'energy', character, 30)
    goal = Goal('Recover energy', character, character, 1, 'recover',
        {False: [condition]}, None, None, None, None, None, None, [],
        graph, 'Basic Needs', {'energy': 30})
    expensive = Action('ExpensiveRecovery', [],
        [{'attribute': 'energy', 'change_value': 10}], cost=5, graph_manager=graph)
    cheap = Action('CheapRecovery', [],
        [{'attribute': 'energy', 'change_value': 10}], cost=1, graph_manager=graph)
    state = State({'energy': 20})
    assert state.get('energy') < 30
    plan = planner.plan_actions(character, goal, state, [expensive, cheap])
    print(json.dumps({'initial_energy': 20, 'target_energy': 30,
        'input_order': [expensive.name, cheap.name],
        'input_costs': [expensive.cost, cheap.cost],
        'actual_plan': None if plan is None else [a.name for a in plan]}), flush=True)
    assert plan, 'Equal-effect cost comparison requires a plan to an unsatisfied target'
    assert plan == [cheap] and sum(a.cost for a in plan) == 1
    final_state = State(state.dict_or_obj.copy())
    for selected in plan:
        final_state = selected.apply_effects(final_state)
    assert final_state.get('energy') >= 30
    assert planner._goal_satisfied(goal, final_state) is True

finally:
    controller.close()
