"""Required #736 contract; run from repository root with python < this file.

This is intentionally retained outside test discovery because the required
contract is unresolved. A failing run is a blocker, never passing coverage.
"""
import inspect
import json
import logging
import random
from unittest.mock import patch
logging.basicConfig(level=logging.ERROR)
random.seed(736)
from actions import Action, Condition, State
from tiny_characters import Character, Goal
from tiny_gameplay_controller import GameplayController
from tiny_goap_system import GOAPPlanner, Plan
from tiny_graph_manager import GraphManager
from networkx import MultiDiGraph

graph = GraphManager()
controller = GameplayController(graph, {
    'headless': True, 'memory': {'mode': 'basic'},
    'characters': {'count': 2, 'use_llm': False, 'use_custom': True,
        'character_data': [{'name': 'ContractActor', 'age': 30,
            'job': 'unemployed', 'energy': 20},
            {'name': 'ContractNeighbor', 'age': 31, 'job': 'unemployed', 'energy': 20}]},
    'checkpoint': {'auto_enabled': False}})
try:
    character = next(c for c in controller.characters.values() if c.name == 'ContractActor')
    assert isinstance(character, Character) and character._initialized
    assert isinstance(graph.G, MultiDiGraph)
    assert graph.characters[character.name] is character
    assert any(node is character for node in graph.G)
    planner = GOAPPlanner(graph)
    action = Action('GraphRecovery', [],
        [{'targets': ['initiator'], 'attribute': 'energy', 'change_value': 10}],
        cost=1, initiator=character, graph_manager=graph)
    # Fixture edges and interaction data use actual registered domain objects.
    neighbor = next(c for c in controller.characters.values() if c.name == 'ContractNeighbor')
    neighbor.possible_interactions = [action]
    graph.G.add_edge(character, neighbor, type='available_interaction')
    assert graph.G.has_edge(character, neighbor)
    assert neighbor.get_possible_interactions() == [action]
    with patch.object(graph, 'get_possible_actions', wraps=graph.get_possible_actions) as observed:
        retrieved = planner.get_available_actions(character)
        observed.assert_called_once_with(character.name)
    print(json.dumps({'registered_interaction': action.name,
        'registered_cost': action.cost, 'retrieved_names': [a.name for a in retrieved],
        'retrieved_effects': [a.effects for a in retrieved]}), flush=True)
    assert [a.name for a in retrieved] == [action.name], 'Graph action retrieval must preserve interaction provenance'
    assert retrieved[0].effects == action.effects and retrieved[0].cost == action.cost

finally:
    controller.close()
