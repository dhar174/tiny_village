from pathlib import Path

class WouldWrite(RuntimeError): pass

def module(version, target):
 source=(OUT/'publish.py').read_text() if version=='fixed' else subprocess.check_output(['git','show',BASE+':docs/planning/issue-734-publication/publish.py'],cwd=ROOT,text=True)
 mod=types.ModuleType('publication');mod.__file__=str(OUT/'publish.py');exec(compile(source,'publish.py','exec'),mod.__dict__)
 mod.OUT=target
 # Bundle integrity is verified separately; the probe exercises only the gate.
 mod.subprocess=types.SimpleNamespace(run=lambda *a,**k:None, DEVNULL=subprocess.DEVNULL)
 return mod

def change(issues, kind):
 issue=issues[11]
 if kind=='state':issue['state']='CLOSED';issue['stateReason']='COMPLETED'
 elif kind=='reason':issue['stateReason']='NOT_PLANNED'
 elif kind=='assignees':issue['assignees']['nodes']=[{'login':'new-owner'}]
 elif kind=='parent':issue['parent']={'number':14}
 elif kind=='dependencies':issue['blockedBy']['nodes']=[]
 elif kind=='timestamp':issue['updatedAt']='2099-01-01T00:00:00Z'
 elif kind=='body':issue['body']+='new content'
 else:raise ValueError(kind)

def publisher(version, kind, phase='preflight', resume=False):
 with tempfile.TemporaryDirectory() as directory:
  target=Path(directory);p=module(version,target);live={int(n):copy.deepcopy(i)for n,i in before.items()};writes=[]
  if resume:
   live[11]=copy.deepcopy(after['11'])
   state={k:copy.deepcopy(v)for k,v in checkpoint.items()if k not in ('annotations','parents','closures','followups','edges')}
   state.update(annotations={'11':copy.deepcopy(checkpoint['annotations']['11'])},parents={},closures={},followups={},edges={})
   if version=='fixed':state['observed_metadata']={'11':p.metadata(live[11])}
   (target/'checkpoint.json').write_text(json.dumps(state))
  if kind and phase=='preflight':change(live,kind)
  p.all_issues=lambda:copy.deepcopy(live)
  def read(numbers):
   found={n:copy.deepcopy(live[n])for n in numbers}
   if kind and phase=='batch':change(found,kind)
   return found
  p.read=read
  def api(path,payload=None,method=None):
   writes.append(path);raise WouldWrite('stopped at first controlled mutation')
  p.api=api
  failure=None
  with contextlib.redirect_stdout(io.StringIO()):
   try:p.run(True)
   except (RuntimeError,AssertionError,SystemExit)as e:failure=type(e).__name__+': '+str(e)
  return {'scenario':f'publication/{phase}/{"resume"if resume else "fresh"}/{kind or "valid"}',
          'version':version,'write_attempts':len(writes),'failure':failure}
